@RestController
@RequestMapping("/api/enrollments")
@RequiredArgsConstructor
public class EnrollmentController {

private final EnrollmentService enrollmentService;

// POST /api/enrollments?userId=1&courseId=2
@PostMapping
public ResponseEntity<ApiResponse<Enrollment>> enroll(
@RequestParam Long userId,
@RequestParam Long courseId) {
return ResponseEntity.status(201)
.body(ApiResponse.ok("Enrolled successfully", enrollmentService.enroll(userId, courseId)));
}

// GET /ap  i/enrollments/user/1
@GetMapping("/user/{userId}")
public ResponseEntity<ApiResponse<List<Enrollment>>> getByUser(@PathVariable Long userId) {
return ResponseEntity.ok(ApiResponse.ok("Enrollments fetched", enrollmentService.getByUser(userId)));
}

// GET /api/enrollments/course/1
@GetMapping("/course/{courseId}")
public ResponseEntity<ApiResponse<List<Enrollment>>> getByCourse(@PathVariable Long courseId) {
return ResponseEntity.ok(ApiResponse.ok("Enrollments fetched", enrollmentService.getByCourse(courseId)));
}

// PATCH /api/enrollments/1/progress?progress=50
@PatchMapping("/{id}/progress")
public ResponseEntity<ApiResponse<Enrollment>> updateProgress(
@PathVariable Long id,
@RequestParam int progress) {
return ResponseEntity.ok(ApiResponse.ok("Progress updated", enrollmentService.updateProgress(id, progress)));
}
}
